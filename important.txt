//selection sort
for(i=0;i<n;i++){
    min = i;
    for(j=i+1;j<n;j++){
        if(arr[min]<arr[j])
            min = j;
    }

    t = arr[min];
    arr[min] = arr[i];
    arr[i] = t;
}

//insertion sort
for(i=1;i<n;i++){
    j = i-1;
    key = arr[i];
    while(j>=0 && key<arr[j]){
        arr[j+1] = arr[j];
        j--
    }
    arr[j+1] = key;
}

//bubble sort
for(i=0;i<n;i++){
    for(j=0;j<n-i-1;j++){
        if(arr[j]>arr[j+1]){
            t = arr[j];
            arr[j] = arr[j+1];
            arr[j+1] = t;
        }
    }
}

//quick sort 2 functions
quicksorlt(low,high){
    pivot=arr[high]
    i = low - 1
    for(j=low; j<high; j++){
        if(arr[j]<=pivot){
            i++
            t = arr[j];
            arr[j] = arr[i];
            arr[i] = t;
        }
    }

    t = arr[i+1];
    arr[i+1] = arr[high];
    arr[high] = t;
    return i+1;
}

splitarray(low, high){
    if(low<high){
        pivot index = quicksort(low,high);
        splitarray(low,pivot index);
        splitarray(pivot index + 1, high);
    }
}
